%   \qty      physics and siunitx both define it.  Here  \qty{1.5}{\meter}  and

%             font (first installed of DejaVu Sans / Segoe UI Symbol / FreeSerif).  Stacked
%             super/subscripts are gathered into ONE script (⁻¹ -> ^{-1}, ₁₂ -> _{12}), also in

% --- helpers for the entries below (they only define macros; nothing is typeset here) -------
% Stacked scripts: a run such as ⁻¹ or ₁₂ is gathered into ONE script, ^{-1} / _{12}.  Written
% one by one they would be ^{-}^{1}, which is a "Double superscript" error.  Each script char
% looks at the next token; if that is another script char of the same kind it keeps gathering.
% The macros are \protected, so headings / table of contents / captions carry the characters
% through unchanged and the run is gathered again when they are re-read.
\def\uni@issup{\let\uni@r\@secondoftwo}      % each registered char appends a test to these
\def\uni@issub{\let\uni@r\@secondoftwo}
\newcommand{\uni@reg}[2]{%                     % #1 = test macro, #2 = the char (already active)
  \begingroup\lccode`\~=`#2 \lowercase{\endgroup
    \g@addto@macro#1{\ifx\uni@nx~\let\uni@r\@firstoftwo\fi}}}
\g@addto@macro\uni@issup{\ifx\uni@nx\uni@sup\let\uni@r\@firstoftwo\fi}   % re-read from .toc
\g@addto@macro\uni@issub{\ifx\uni@nx\uni@sub\let\uni@r\@firstoftwo\fi}
\protected\def\uni@sup#1{%
  \ifdefined\uni@acc\edef\uni@acc{\uni@acc#1}\else\def\uni@acc{#1}\fi
  \futurelet\uni@nx\uni@supchk}
\protected\def\uni@sub#1{%
  \ifdefined\uni@acc\edef\uni@acc{\uni@acc#1}\else\def\uni@acc{#1}\fi
  \futurelet\uni@nx\uni@subchk}
\def\uni@supchk{\uni@issup\uni@r{}{\uni@supend}}
\def\uni@subchk{\uni@issub\uni@r{}{\uni@subend}}
\def\uni@supend{\expandafter\uni@supout\expandafter{\uni@acc}\let\uni@acc\@undefined}
\def\uni@subend{\expandafter\uni@subout\expandafter{\uni@acc}\let\uni@acc\@undefined}
\def\uni@supout#1{\ensuremath{^{#1}}}
\def\uni@subout#1{\ensuremath{_{#1}}}
\newcommand{\uni@defsup}[2]{\newunicodechar{#1}{\uni@sup{#2}}\uni@reg\uni@issup{#1}}
\newcommand{\uni@defsub}[2]{\newunicodechar{#1}{\uni@sub{#2}}\uni@reg\uni@issub{#1}}
% Shapes / dingbats: \uni@sym{<hex code point>} prints the glyph from the symbol fallback font.
% (\char is used because the character itself is active and would call itself again.)
\newcommand{\uni@sym}[1]{\mbox{\unisymfont\char"#1\relax}}
