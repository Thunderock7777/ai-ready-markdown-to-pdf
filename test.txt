```python

import os
import random

# Determine the absolute path of the directory containing this script
script_dir = os.path.dirname(os.path.abspath(__file__))
file_path = os.path.join(script_dir, 'input.txt')

try:
    with open(file_path, 'r', encoding='utf-8') as file:
        # Read all lines and remove empty ones
        lines = [line.strip() for line in file.readlines() if line.strip()]
        
    if not lines:
        print("The file 'input.txt' is empty.")
    else:
        # Pick a random line and reverse it
        chosen_line = random.choice(lines)
        print(f"Random line selected: {chosen_line}")
        print(f"Reversed text: {chosen_line[::-1]}")
        
except FileNotFoundError:
    print(f"Error: Make sure 'input.txt' is located at {script_dir}")
```
